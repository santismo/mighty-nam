cmake_minimum_required(VERSION 3.20)
project(mighty_plug_nam_lab LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ENABLE_SANITIZERS "Offline host address/undefined-behavior checks" OFF)
if(ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
file(GLOB NAM_SOURCES CONFIGURE_DEPENDS third_party/nam/NAM/*.cpp)
add_library(nam_core OBJECT ${NAM_SOURCES})
target_include_directories(nam_core PUBLIC third_party/nam third_party/eigen third_party/json)
target_compile_definitions(nam_core PUBLIC NAM_SAMPLE_FLOAT)
add_library(mpp_core src/engine.cpp)
target_include_directories(mpp_core PUBLIC include)
target_compile_options(mpp_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mpp_lab src/lab.cpp)
target_link_libraries(mpp_lab PRIVATE mpp_core nam_core)
add_executable(mpp_tests tests/test_engine.cpp target/audio_module.cpp)
target_link_libraries(mpp_tests PRIVATE mpp_core nam_core)
enable_testing()
add_test(NAME dsp_properties COMMAND mpp_tests ${CMAKE_CURRENT_SOURCE_DIR}/models)
add_library(mpp_board OBJECT target/board/audio_transport.cpp target/board/codec_control.cpp target/board/startup_memory.cpp)
target_include_directories(mpp_board PUBLIC target/board include)
target_compile_options(mpp_board PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mpp_board_tests tests/test_board.cpp target/audio_module.cpp)
target_link_libraries(mpp_board_tests PRIVATE mpp_board mpp_core)
add_test(NAME board_transport_and_startup COMMAND mpp_board_tests)
add_executable(mpp_hybrid_tests tests/test_hybrid.cpp target/hybrid/amp_hook.cpp)
target_include_directories(mpp_hybrid_tests PRIVATE target)
target_link_libraries(mpp_hybrid_tests PRIVATE mpp_core)
add_test(NAME hybrid_amp_adapter COMMAND mpp_hybrid_tests)
add_library(mpp_standalone OBJECT target/standalone/prototype.cpp target/standalone/clock_plan.cpp target/a2/a2_lite.cpp)
target_compile_options(mpp_standalone PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(mpp_standalone_tests tests/test_standalone.cpp)
target_link_libraries(mpp_standalone_tests PRIVATE mpp_standalone mpp_board)
add_test(NAME standalone_startup_audio_faults COMMAND mpp_standalone_tests)
add_library(mpp_hardware_backend OBJECT target/hardware/rt1176_backend.cpp target/hardware/handoff.cpp)
target_compile_options(mpp_hardware_backend PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mpp_hardware_tests tests/test_hardware_backend.cpp)
target_link_libraries(mpp_hardware_tests PRIVATE mpp_hardware_backend mpp_standalone mpp_board)
add_test(NAME hardware_register_backend_faults COMMAND mpp_hardware_tests)

add_library(mpp_model_receiver OBJECT target/link/model_receiver.cpp)
target_compile_options(mpp_model_receiver PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mpp_model_receiver_tests tests/test_model_receiver.cpp)
target_link_libraries(mpp_model_receiver_tests PRIVATE mpp_model_receiver mpp_standalone mpp_board)
add_test(NAME model_transfer_faults COMMAND mpp_model_receiver_tests)
add_executable(mpp_sysex_codec_tests tests/test_sysex_codec.cpp target/link/sysex_codec.cpp)
target_compile_options(mpp_sysex_codec_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME model_sysex_bounds COMMAND mpp_sysex_codec_tests)

add_executable(mpp_startup_clock_tests tests/test_startup_clock.cpp target/hardware/startup_clock.cpp)
target_compile_options(mpp_startup_clock_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME startup_clock_read_only COMMAND mpp_startup_clock_tests)

add_executable(mpp_probe_wire_tests tests/test_probe_wire.c)
target_compile_options(mpp_probe_wire_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME silent_probe_wire COMMAND mpp_probe_wire_tests)

add_library(mpp_program OBJECT src/program.cpp)
target_include_directories(mpp_program PUBLIC include)
target_compile_options(mpp_program PRIVATE -Wall -Wextra -Wpedantic)

add_executable(mpp_program_tests tests/test_program.cpp target/a2/a2_lite.cpp)
target_link_libraries(mpp_program_tests PRIVATE mpp_program mpp_core)
target_compile_options(mpp_program_tests PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
set(NAM_CHAIN_PROGRAM "" CACHE FILEPATH "App-exported NPC1 chain for integration testing")
set(NAM_A2_MODEL "" CACHE FILEPATH "Locally licensed A2 Lite .nam file for integration testing")
if(NAM_CHAIN_PROGRAM AND NAM_A2_MODEL)
  add_test(NAME app_compiled_a2_effects_chain COMMAND mpp_program_tests ${NAM_CHAIN_PROGRAM} ${NAM_A2_MODEL})
endif()
